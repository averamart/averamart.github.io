অনলাইন দোকানের Firebase Rules (সংস্করণ ৫.১২) — আগের ৩টি দোকান-ব্লক মুছে নিচের ৪টি বসান
=====================================================================================
আগে যে ৩টি ব্লক (averamart_public, averamart_public_img, averamart_weborders) বসিয়েছিলেন সেগুলো মুছে ফেলুন,
তারপর নিচের পুরোটা (৪টি ব্লক) শেষের দুটো `}`-এর ওপরে বসিয়ে Publish করুন। নতুন যোগ: ছাড়ের তথ্য (discount) ও অর্ডার ট্র্যাক (averamart_track)।

    // ---- অনলাইন দোকান ----
    match /averamart_public/{d} {
      allow read: if true;
      allow write: if hasRole();
    }
    match /averamart_public_img/{d} {
      allow read: if true;
      allow write: if hasRole();
    }
    match /averamart_weborders/{id} {
      allow create: if request.resource.data.keys().hasOnly(['customerName','phone','address','note','area','items','discount','subtotal','delivery','deliveryTbd','total','status','source','createdAt'])
        && request.resource.data.keys().hasAll(['customerName','phone','address','items','total','status','createdAt'])
        && request.resource.data.customerName is string && request.resource.data.customerName.size() > 0 && request.resource.data.customerName.size() <= 80
        && request.resource.data.phone is string && request.resource.data.phone.matches('^01[0-9]{9}$')
        && request.resource.data.address is string && request.resource.data.address.size() >= 8 && request.resource.data.address.size() <= 300
        && (!('note' in request.resource.data) || (request.resource.data.note is string && request.resource.data.note.size() <= 300))
        && request.resource.data.items is list && request.resource.data.items.size() > 0 && request.resource.data.items.size() <= 40
        && request.resource.data.total is number && request.resource.data.total >= 0 && request.resource.data.total < 1000000
        && request.resource.data.status == 'new'
        && request.resource.data.createdAt == request.time;
      allow read: if hasRole();
      allow update: if hasRole() && resource.data.status == 'new'
        && request.resource.data.status in ['imported','cancelled']
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','orderId','handledBy','handledAt']);
      allow delete: if false;
    }
    // অর্ডার ট্র্যাকিং: শুধু অবস্থা। কাস্টমার নম্বর জানলে একটি একটি করে দেখতে পারে (তালিকা দেখা যায় না)
    match /averamart_track/{id} {
      allow get: if true;
      allow list: if false;
      allow create: if (request.resource.data.keys().hasOnly(['s','t'])
          && request.resource.data.s == 'new'
          && request.resource.data.t == request.time
          && existsAfter(/databases/$(database)/documents/averamart_weborders/$(id)))
        || hasRole();
      allow update: if hasRole();
      allow delete: if false;
    }

জরুরি: ট্র্যাক ব্লক না বসালে কাস্টমারের অর্ডারই যাবে না (কারণ অর্ডারের সাথে ট্র্যাক ডকও একসাথে লেখা হয়)। তাই Rules বসানোর আগে নতুন ফাইল আপলোড করবেন না।
